cmake_minimum_required(VERSION 3.20)
project(LaboratorioOrdenamiento LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(algoritmos src/Algoritmos.cpp src/Lista.cpp)
target_include_directories(algoritmos PUBLIC src)
if(MSVC)
  target_compile_options(algoritmos PRIVATE /W4 /utf-8)
endif()
add_executable(LaboratorioOrdenamiento src/main.cpp)
target_link_libraries(LaboratorioOrdenamiento PRIVATE algoritmos)
if(MSVC)
  target_compile_options(LaboratorioOrdenamiento PRIVATE /W4 /utf-8)
  target_link_options(LaboratorioOrdenamiento PRIVATE /STACK:33554432)
endif()
enable_testing()
add_executable(Pruebas tests/pruebas.cpp)
target_link_libraries(Pruebas PRIVATE algoritmos)
if(MSVC)
  target_compile_options(Pruebas PRIVATE /W4 /utf-8)
  target_link_options(Pruebas PRIVATE /STACK:33554432)
endif()
add_test(NAME AlgoritmosYEntrada COMMAND Pruebas)
